## Appendix: the script

Python 3, no packages. It imports the calculator from `docs/scale-sim/` and changes its enemy brain in memory, in a
separate process per case. Run: `python3 brains.py 24 [run2|run3|run4|run5|run6]` (24 fights per seed set). The
first run is the main table; run2 to run6 are the room-wrecked side, the new-player crews, the full fire chain, the
x5 pieces and the chases. 18,624 fights took about 15 minutes on 8 cores.

Words in the script: `scan30` = A2, `chain` / `chain2` = A1 (one shot per lock / one shot per arming), `heard` = A3,
`pdc` = A5, `band` = B4, `fresh` / `freshroom` / `guard` = B1, `arm8` = an 8 s crew.

Known limits: the Garuda's power isn't run (its charge assumes WEP 4); its own scan costs it nothing and isn't
heard; A4 isn't modelled; after a hit it guesses a spot 2,000 out along the side that was hit.

<details><summary>brains.py</summary>

```python
"""Enemy brain variants on Wolf Hunt, played by Player 1's calculator (docs/scale-sim/scalesim.py), unchanged.
Each variant patches the calculator's Brain from outside, in its own process. Nothing on disk is changed.
Run: python3 brains.py [fights per seed set, default 24]"""
import sys, math, multiprocessing as mp

SIM = '/Users/Shagun/Documents/GitHub/Killshot/docs/scale-sim'

def patch(S, parts):
    B, Ship = S.Brain, S.Ship
    o_slug, o_he, o_run = Ship.take_slug, Ship.take_he, Ship.take_run
    def mark(s, side):
        if s.name == 'Garuda':
            if not hasattr(s, 'first_side'): s.first_side = side
            s.struck = side
    def take_slug(s, side, by): mark(s, side); return o_slug(s, side, by)
    def take_he(s, side, by): mark(s, side); return o_he(s, side, by)
    def take_run(s, side, disable, arm, comp): mark(s, 2 if disable else side); return o_run(s, side, disable, arm, comp)
    Ship.take_slug, Ship.take_he, Ship.take_run = take_slug, take_he, take_run

    o_think, o_listen, o_steer, o_side = B.think, B.listen, B.steer, B.side_on

    def think(b, us, fighters, sim, now):
        s = b.s
        if not hasattr(b, 'bars'):
            b.bars, b.fix_since, b.prev_fix, b.last_t, b.vel, b.vel_at, b.sideon = {}, 1e9, False, now, (0.0, 0.0), -1e9, False
            b.charge, b.last_shot = 100.0, {}
        b.charge = min(100.0, b.charge + 4 * s.wp['wepFeedPerUnit'] * (now - b.last_t))   # its capacitor at WEP 4, as ours
        if 'heard' in parts:      # only a real strike counts as a hit (not fire, not a bump)
            struck = getattr(s, 'struck', None); s.struck = None; b._struck = struck
            if b.health >= 0: b.health = s.hp() + (1 if struck is not None else 0)
        o_think(b, us, fighters, sim, now)
        fix = b.level == S.ALERT and b.has_fix(now)
        if fix and not b.prev_fix: b.fix_since = now
        if fix and ('scan30' in parts) and now - b.vel_at >= 30: b.vel, b.vel_at = (us.vx, us.vy), now   # its own scan, every 30 s
        b.prev_fix = fix
        dt = now - b.last_t; b.last_t = now
        rel_us = (S.brg(us.x - s.x, us.y - s.y) - s.hdg) % 360
        for mid, m in s.mounts.items():       # the gun bar turns at 3 deg/s toward us inside its arc (WeaponSystem rule)
            if m['kind'] != 'RG': continue
            arc = int(m['arc']); want = rel_us if S.angdiff(rel_us, arc) <= m['halfArc'] else arc
            bar = b.bars.get(mid, float(arc)); d = S.delta(bar, want); step = s.wp['gunTurnDegPerSecond'] * dt
            b.bars[mid] = (want if abs(d) <= step else bar + math.copysign(step, d)) % 360

    def listen(b, us, fighters, dist, now, hit):
        o_listen(b, us, fighters, dist, now, hit)
        st = getattr(b, '_struck', None)
        if 'heard' in parts and hit and st is not None and not b.hears_us and b.last_heard == now and b.last_known not in {(f.x, f.y) for f in fighters}:
            a = math.radians(b.s.hdg + 90 * st + b.rnd.uniform(-45, 45))     # it knows the face that was hit, not the range
            b.last_known = (b.s.x + math.sin(a) * 2000, b.s.y + math.cos(a) * 2000)

    def shoot(b, us, mid, m, dist, now, sim):
        s = b.s
        if s.ammo[mid] <= 0 or now < b.next_shot.get(mid, 0) or dist > m['rangeUnits'] or mid not in s.dots: return
        if 'chain' in parts:      # lock fills 10 s, holds 10 s; its crew arms in ARM s; fire only on the hold, down the bar
            ph = now - b.fix_since
            if ph < 0 or (ph % 20.0) < 10 + ARM: return
        if 'chain2' in parts:     # our chain: lock fills 10 s, holds 10 s; one shot per arming; a shot costs 70 charge
            ph = now - b.fix_since
            if ph < 0 or (ph % 20.0) < 10: return
            hold = now - ((ph % 20.0) - 10)
            if now < max(hold, b.last_shot.get(mid, -1e9)) + ARM or b.charge < 70: return
        if 'nolead' in parts: vx, vy = 0.0, 0.0
        elif 'scan30' in parts: vx, vy = b.vel
        else: vx, vy = us.vx, us.vy
        t = S.intercept(us.x - s.x, us.y - s.y, vx, vy, s.wp['slugSpeedUnits'])
        lead = S.brg(us.x - s.x + vx * t, us.y - s.y + vy * t) if t else S.brg(us.x - s.x, us.y - s.y)
        rel = S.jsround((lead - s.hdg) % 360) % 360
        if S.angdiff(rel, m['arc']) > m['halfArc']: return
        if ('chain' in parts or 'chain2' in parts) and S.angdiff(b.bars.get(mid, m['arc']), rel) > 1: return
        gx, gy = s.gun_at(mid)
        r = S.Round(gx, gy, lead + (b.rnd.random() * 2 - 1) * b.c['aimErrorDeg'], s.wp['slugSpeedUnits'], m['rangeUnits'], s.wp, shooter=s, fired=now,
                    cf=(us.x, us.y, us.vx, us.vy, us.hdg))
        sim.incoming.append(r); s.ammo[mid] -= 1; sim.their_shots += 1; sim.event(now, 'slug in')
        b.next_shot[mid] = now + b.c['railgunReloadSeconds'] + b.rnd.random() * b.c['railgunJitterSeconds']
        if 'chain2' in parts: b.next_shot[mid] = now; b.last_shot[mid] = now; b.charge -= 70

    def steer(b, us, dist, now, fix, sim):
        if 'band' in parts: b.c['fightAtShare'] = 0.95 if getattr(b, 'sideon', False) else 0.8
        o_steer(b, us, dist, now, fix, sim)
        if 'band' in parts: b.sideon = fix and dist <= b.c['fightAtShare'] * b.reach()

    def side_on(b, bb):
        if 'fresh' not in parts and 'guard' not in parts and 'freshroom' not in parts: return o_side(b, bb)
        c = []
        for mid, m in b.s.mounts.items():
            if m['kind'] != 'RG' or b.s.ammo[mid] <= 0: continue
            h = (bb - int(m['arc'])) % 360; face = round(int(m['arc']) / 90) % 4
            c.append((h, S.angdiff(h, b.s.hdg), face))
        if not c: return bb
        if 'guard' in parts:      # keep the reactor face (PORT) away while the other gun has slugs
            keep = [x for x in c if x[2] != 3]; c = keep or c
        elif 'freshroom' in parts:  # turn the other face once the room behind the shown face is wrecked
            shown = min(c, key=lambda x: x[1]); room = lambda f: b.s.health(S.BEHIND[f])
            if room(shown[2]) <= 0:
                fresh = [x for x in c if room(x[2]) > 0]; c = fresh or c
        else:                     # turn the other face once the shown face's armour is gone
            shown = min(c, key=lambda x: x[1])
            if b.s.armour[shown[2]] <= 0:
                fresh = [x for x in c if b.s.armour[x[2]] > 0]; c = fresh or c
        return min(c, key=lambda x: x[1])[0]

    B.think, B.listen, B.shoot, B.steer, B.side_on = think, listen, shoot, steer, side_on

    if 'pdc' in parts:            # its PDC group A (starboard half, 600, 70%) gets one try at each of our torpedoes
        o_torp = S.torp_step
        def torp_step(r, dt, target):
            if target is not None and target.name == 'Garuda' and not (r.tried & 16) and not target.dead:
                d = math.hypot(r.x - target.x, r.y - target.y)
                for mid, m in target.mounts.items():
                    if m['kind'] != 'PD' or target.ammo[mid] < target.wp['pdcRoundsPerGo']: continue
                    gx, gy = target.gun_at(mid); rel = S.jsround((S.brg(r.x - gx, r.y - gy) - target.hdg) % 360) % 360
                    if math.hypot(r.x - gx, r.y - gy) <= m['rangeUnits'] and S.angdiff(rel, int(m['arc'])) <= m['halfArc']:
                        r.tried |= 16; target.ammo[mid] -= int(target.wp['pdcRoundsPerGo'])
                        if target.rnd.random() < target.wp['pdcKillChance']: return 'miss'
                        break
            return o_torp(r, dt, target)
        S.torp_step = torp_step
```
